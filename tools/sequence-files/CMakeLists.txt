cmake_minimum_required(VERSION 3.20)
project(sc55_sequence_files LANGUAGES CXX)
set(SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../Plugins/Source")

    ${SOURCE}/MidiFilePlayer.cpp ${SOURCE}/RcpFilePlayer.cpp ${SOURCE}/WrdFilePlayer.cpp)
